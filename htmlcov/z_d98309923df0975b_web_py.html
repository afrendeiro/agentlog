77def page(records: dict[str, dict], days_back: int = 90) -> str: 

78 rows = sorted(records.values(), key=lambda r: r["end"], reverse=True) 

79 per_day: dict[str, float] = defaultdict(float) 

80 per_proj: dict[str, list[dict]] = defaultdict(list) 

81 for r in rows: 

82 for d, m in r["days"].items(): 

83 per_day[d] += m 

84 per_proj[r["project"]].append(r) 

85 today = date.today() 

86 week = sum(m for d, m in per_day.items() if d >= (today - timedelta(days=6)).isoformat()) 

87 

88 # Activity strip: one bar per day, linking to that day's section. 

89 span = [(today - timedelta(days=i)).isoformat() for i in range(days_back - 1, -1, -1)]

92 f'<a href="#d{d}" class="{"zero" if not per_day[d] else ""}" style="height:{max(per_day[d] / peak * 100, 2):.0f}%"'

134def entry(r: dict) -> str: 

135 start, end = datetime.fromisoformat(r["start"]), datetime.fromisoformat(r["end"]) 

136 when = f"{start:%H:%M}–{end:%H:%M}" if start.date() == end.date() else f"{start:%d %b %H:%M} – {end:%d %b %H:%M}" 

137 s = r.get("summary") or {} 

138 title = s.get("title") or r.get("title") or r["first_prompt"][:100] 

139 fields = [(k, s[k]) for k in ("before", "did", "after") if s.get(k)] or ( 

140 [("after", r["recap"])] if r.get("recap") else [] 

141 ) 

142 dl = "".join(f"<dt>{k}</dt><dd>{escape(v)}</dd>" for k, v in fields) 

143 extra = " · ".join( 

144 x 

145 for x in (r.get("model") or "", f"{r['prompts']} prompts", f"${r['cost_usd']:.2f}" if r.get("cost_usd") else "") 

146 if x 

147 ) 

148 tip = escape(f"{r['key']}\n{r['cwd']}\nfirst prompt: {r['first_prompt']}")

158class Handler(BaseHTTPRequestHandler): 

159 def do_GET(self) -> None: 

160 if self.path.split("?")[0] == "/api/sessions": 

161 body, ctype = json.dumps(list(store.load().values())).encode(), "application/json" 

162 elif self.path.split("?")[0] in ("/", "/index.html"): 

163 body, ctype = page(store.load()).encode(), "text/html; charset=utf-8" 

164 else: 

165 self.send_error(404) 

166 return 

167 self.send_response(200) 

168 self.send_header("Content-Type", ctype) 

169 self.send_header("Content-Length", str(len(body))) 

170 self.end_headers() 

171 self.wfile.write(body)
